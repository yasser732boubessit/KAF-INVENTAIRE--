enum UserRole {
  TEAM_1   // Desktop Offline / Asset creation
  TEAM_2   // Web Portal / Physical verification
  TEAM_3   // Mobile PWA / Loading & Registry
  ADMIN    // Super Admin / All permissions + ETL
}

enum WorkflowState {
  AJOUTE
  SYNCHRONISE
  VERIFIE
  CHARGE
  REJETE
}

enum PhysicalState {
  ACTIF
  EN_PANNE
  EN_MAINTENANCE
  EN_STOCK
  HORS_SERVICE
  REFORME
}

enum InventoryStatus {
  CONFORME
  ECART_LOCALISATION
  MANQUANT
  NON_VERIFIE
  NON_ENREGISTRE
}

enum AssetCategory {
  Mechanical
  Electrical
  Automation
  Instrumentation
  Safety
}

// -------------------------------------------------------------
// Models

model User {
  id           String        @id @default(uuid())
  username     String        @unique
  fullName     String
  email        String?       @unique
  passwordHash String
  role         UserRole      @default(TEAM_1)
  isActive     Boolean       @default(true)
  createdAt    DateTime      @default(now())
  updatedAt    DateTime      @updatedAt

  // Relations
  createdAssets        Asset[]              @relation("AssetCreator")
  verifiedRecords      VerificationRecord[] @relation("VerifierUser")
  loadingRecords       LoadingRecord[]      @relation("LoaderUser")
  auditLogs            AuditLog[]           @relation("UserAudits")

  @@map("users")
}

model Asset {
  id                   String          @id @default(uuid())
  officialCode         String          @unique
  name                 String
  nameFr               String?
  nameAr               String?
  sku                  String?
  serialNumber         String          @unique
  barcode              String          @unique
  rfidTag              String?         @unique
  category             AssetCategory   @default(Mechanical)
  zone                 String          @default("ZONE-A")
  expectedBayLocation  String
  scannedBayLocation   String?
  inventoryStatus      InventoryStatus @default(NON_VERIFIE)
  workflowState        WorkflowState   @default(AJOUTE)
  physicalState        PhysicalState   @default(ACTIF)
  condition            String          @default("nominal")
  tamperSealIntact     Boolean         @default(true)
  rfidVerified         Boolean         @default(false)
  serialVerified       Boolean         @default(false)
  weightKg             Float           @default(0.0)
  notes                String?
  notesFr              String?
  imageUrl             String?
  version              Int             @default(1)

  // Tracking timestamps & authors
  createdAt            DateTime        @default(now())
  updatedAt            DateTime        @updatedAt
  createdById          String?
  createdBy            User?           @relation("AssetCreator", fields: [createdById], references: [id])
  lastScannedAt        DateTime?
  scannedBy            String?

  syncBatchId          String?
  syncBatch            SyncBatch?      @relation(fields: [syncBatchId], references: [id])

  // Verification & Loading links
  verificationRecords  VerificationRecord[]
  loadingRecords       LoadingRecord[]
  auditLogs            AuditLog[]

  @@index([workflowState])
  @@index([barcode])
  @@index([serialNumber])
  @@index([officialCode])
  @@map("assets")
}

model SyncBatch {
  id              String        @id @default(uuid())
  deviceId        String
  receivedAt      DateTime      @default(now())
  clientTimestamp DateTime?
  batchSize       Int           @default(0)
  insertedCount   Int           @default(0)
  updatedCount    Int           @default(0)
  conflictCount   Int           @default(0)
  status          String        @default("SUCCESS") // SUCCESS, PARTIAL, REJECTED
  assets          Asset[]
  auditLogs       AuditLog[]

  @@map("sync_batches")
}

model VerificationRecord {
  id                  String        @id @default(uuid())
  assetId             String
  asset               Asset         @relation(fields: [assetId], references: [id], onDelete: Cascade)
  verifiedByUserId    String
  verifiedByUser      User          @relation("VerifierUser", fields: [verifiedByUserId], references: [id])
  verifiedAt          DateTime      @default(now())
  bayLocation         String
  condition           String
  tamperSealIntact    Boolean
  rfidMatches         Boolean
  serialMatches       Boolean
  notes               String?
  resultStatus        String        @default("CONFORME") // CONFORME, ECART, ANOMALIE

  @@index([assetId])
  @@map("verification_records")
}

model LoadingRecord {
  id                  String        @id @default(uuid())
  assetId             String
  asset               Asset         @relation(fields: [assetId], references: [id], onDelete: Cascade)
  loadedByUserId      String
  loadedByUser        User          @relation("LoaderUser", fields: [loadedByUserId], references: [id])
  loadedAt            DateTime      @default(now())
  destinationBay      String
  vehiclePlate        String?
  containerCode       String?
  notes               String?

  @@index([assetId])
  @@map("loading_records")
}

  id             String        @id @default(uuid())
  action         String
  entityType     String        @default("ASSET")
  assetId        String?
  asset          Asset?        @relation(fields: [assetId], references: [id], onDelete: SetNull)
  userId         String?
  user           User?         @relation("UserAudits", fields: [userId], references: [id], onDelete: SetNull)
  role           String
  previousState  String?
  newState       String?
  details        String?
  ipAddress      String?
  userAgent      String?
  syncBatchId    String?
  syncBatch      SyncBatch?    @relation(fields: [syncBatchId], references: [id], onDelete: SetNull)
  createdAt      DateTime      @default(now())

  @@index([assetId])
  @@index([createdAt])
  @@map("audit_logs")
}

import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ValidationPipe, Logger } from '@nestjs/common';
import helmet from 'helmet';

async function bootstrap() {
  const logger = new Logger('KAF_INVENTAIRE_BOOTSTRAP');
  const app = await NestFactory.create(AppModule);

  // Hardening & Security
  app.use(helmet());

  // CORS for Web Portal, Electron Desktop, and PWA Mobile
  app.enableCors({
    origin: '*',
    methods: 'GET,HEAD,PUT,PATCH,POST,DELETE,OPTIONS',
    credentials: true,
    allowedHeaders: 'Content-Type, Accept, Authorization, x-device-id, x-device-api-key',
  });

  // Global prefix as defined in API Documentation: /api/v1
  app.setGlobalPrefix('api/v1');

  // DTO Validation
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      transform: true,
      forbidNonWhitelisted: false,
    }),
  );

  const port = process.env.PORT || 5000;
  await app.listen(port);
  logger.log(`ðŸš€ KAF INVENTAIRE Central API is running on port: ${port}`);
  logger.log(`ðŸ“¡ Endpoints available under http://localhost:${port}/api/v1`);
  logger.log(`âš™ï¸  Active Database: ${process.env.DATABASE_URL?.split('@')[1] || 'PostgreSQL 16'}`);
}
bootstrap();
